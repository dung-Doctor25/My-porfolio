Step 01 / Pipeline

{% if current_lang == 'en' %}Advanced Data Querying{% else %}Truy vấn dữ liệu chuyên sâu (SQL){% endif %}

{% if current_lang == 'en' %}Monthly Sales Category Probability Analysis{% else %}Phân tích Xác suất Đơn hàng theo Danh mục Sản phẩm từng Tháng{% endif %}

{% if current_lang == 'en' %}Utilizing Common Table Expressions (CTEs) and conditional aggregation to evaluate customer purchasing trends and market penetration.{% else %}Sử dụng các bảng tạm thời (CTEs) và hàm tổng hợp có điều kiện để tính toán tỷ lệ đơn hàng, phục vụ phân tích xu hướng và hành vi tiêu dùng.{% endif %} 

WITH 
    MonthlySales AS (
        SELECT 
            strftime('%m', b.time_created) AS month,
            p.category_id,
            c.category_code,  
            c.category_name,  
            COUNT(DISTINCT b.bill_id) AS unique_orders_per_category  
        FROM bill b
        JOIN bill_line bl ON b.bill_id = bl.bill_id
        JOIN product p ON bl.product_id = p.product_id
        JOIN category c ON p.category_id = c.category_id
        GROUP BY strftime('%m', b.time_created), p.category_id, c.category_code, c.category_name 
    ),
    TotalMonthlyOrders AS (
        SELECT 
            strftime('%m', b.time_created) AS month,  
            COUNT(DISTINCT b.bill_id) AS total_orders 
        FROM bill b
        GROUP BY strftime('%m', b.time_created) 
    ),
    Probabilities AS (
        SELECT 
            ms.month,
            ms.category_code,  
            ms.category_name,  
            ms.unique_orders_per_category,
            tm.total_orders,
            CAST(ms.unique_orders_per_category AS REAL) / tm.total_orders AS sale_probability 
        FROM MonthlySales ms
        JOIN TotalMonthlyOrders tm ON ms.month = tm.month 
    )
SELECT 
    "T" || month AS "Tháng",
    ROUND(MAX(CASE WHEN category_code = 'BOT' THEN sale_probability END) * 100, 1) || "%" AS "[BOT] Bột",
    ROUND(MAX(CASE WHEN category_code = 'SET' THEN sale_probability END) * 100, 1) || "%" AS "[SET] Set trà",
    ROUND(MAX(CASE WHEN category_code = 'THO' THEN sale_probability END) * 100, 1) || "%" AS "[THO] Trà hoa",
    ROUND(MAX(CASE WHEN category_code = 'TMX' THEN sale_probability END) * 100, 1) || "%" AS "[TMX] Trà mix",
    ROUND(MAX(CASE WHEN category_code = 'TTC' THEN sale_probability END) * 100, 1) || "%" AS "[TTC] Trà củ, quả sấy"
FROM Probabilities
GROUP BY month
ORDER BY month;
